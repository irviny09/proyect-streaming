-- -----------------------------------------------------
-- procedure sp_addNewMovie
-- -----------------------------------------------------

DELIMITER $$
USE `db_streaming`$$
CREATE DEFINER=`root`@`localhost` PROCEDURE `sp_addNewMovie`(
	_nombre varchar(50),
    _descripcion TEXT(500),
    _activo bit(1),
    _genero int,
    _url varchar(200)
)
BEGIN

	declare _imagenId int;
    declare _peliculaId int;
	insert into tbl_ope_imagenes 
    values(
		null,
        _nombre,
        _url
    );
    
    set _imagenId = (select ImagenId from tbl_ope_imagenes order by ImagenId desc limit 1);
    
    insert into tbl_ope_peliculas 
    values(
		null,
        _nombre,
        _descripcion,
        _activo,
        _imagenId,
        _genero
    );
    
    set _peliculaId = (select PeliculaId from tbl_ope_peliculas order by PeliculaId desc limit 1);
    
    insert into tbl_ope_logs 
    values(
		null,
        upper(concat("Se ingreso una nueva pelicula con ID: " , _peliculaId , " y imagen con ID: " , _imagenId)),
        now()
    );
    
    
END$$

-- -----------------------------------------------------
-- procedure sp_showAMovieById
-- -----------------------------------------------------

DELIMITER $$
USE `db_streaming`$$
CREATE DEFINER=`root`@`localhost` PROCEDURE `sp_showAMovieById`(
	_peliculaId int
)
BEGIN
	select 
	p.Pelicula_Nombre,
	p.Pelicula_Descripcion,
	p.Pelicula_Activo,
	g.Genero_Genero,
	i.Imagen_Nombre,
	i.Imagen_URL
	from tbl_ope_peliculas p
	inner join tbl_cat_generos g
	on p.Pelicula_GeneroId = g.GeneroId
	inner join tbl_ope_imagenes i
	on p.Pelicula_ImagenId = i.ImagenId
    where p.PeliculaId = _peliculaId;
    
    insert into tbl_ope_logs 
    values(
		null,
        upper(concat("Se mostro una pelicula con ID: " , _peliculaId)),
        now()
    );
END$$

DELIMITER $$
USE `db_streaming`$$

BEGIN

	concat(u.Usuario_Nombre , " " , u.Usuario_ApellidoPat , " " , u.Usuario_ApellidoMat) as Usuario,
